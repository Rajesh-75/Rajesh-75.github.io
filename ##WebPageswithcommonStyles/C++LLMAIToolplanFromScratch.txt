## Step 2 — Train a Custom Tokenizer

C++ has unique tokens (`::`, `->`, `#include`, `std::`, template syntax). A custom tokenizer handles these better.

```python
from tokenizers import ByteLevelBPETokenizer

tokenizer = ByteLevelBPETokenizer()

tokenizer.train(
    files=["cpp_corpus.txt"],
    vocab_size=32000,          # 32K–64K is standard
    min_frequency=2,
    special_tokens=[
        "<s>", "</s>", "<unk>", "<pad>",
        "<|code|>", "<|comment|>",  # domain-specific
        "<|function|>", "<|class|>"
    ]
)

tokenizer.save_model("./cpp-tokenizer")
```

**Why custom tokenizer matters for C++:**
- `std::vector<int>` = 1–2 tokens instead of 6–8
- `#include <iostream>` handled as natural unit
- Template syntax `template<typename T>` tokenized efficiently

### Architecture (modern best practices):

```python
import torch
import torch.nn as nn
import math

class RMSNorm(nn.Module):
    """Better than LayerNorm - used in LLaMA, Mistral"""
    def __init__(self, dim, eps=1e-6):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(dim))
        self.eps = eps

    def forward(self, x):
        rms = x.pow(2).mean(-1, keepdim=True).add(self.eps).rsqrt()
        return x * rms * self.weight


class RotaryEmbedding(nn.Module):
    """RoPE - Rotary Position Embedding (LLaMA-style)"""
    def __init__(self, dim, max_seq_len=4096):
        super().__init__()
        inv_freq = 1.0 / (10000 ** (torch.arange(0, dim, 2).float() / dim))
        self.register_buffer("inv_freq", inv_freq)

    def forward(self, seq_len, device):
        t = torch.arange(seq_len, device=device).float()
        freqs = torch.outer(t, self.inv_freq)
        return torch.cat([freqs, freqs], dim=-1)


class GroupedQueryAttention(nn.Module):
    """GQA - used in Mistral/LLaMA3 for efficiency"""
    def __init__(self, d_model, n_heads, n_kv_heads):
        super().__init__()
        self.n_heads = n_heads
        self.n_kv_heads = n_kv_heads
        self.head_dim = d_model // n_heads

        self.q_proj = nn.Linear(d_model, n_heads * self.head_dim, bias=False)
        self.k_proj = nn.Linear(d_model, n_kv_heads * self.head_dim, bias=False)
        self.v_proj = nn.Linear(d_model, n_kv_heads * self.head_dim, bias=False)
        self.o_proj = nn.Linear(d_model, d_model, bias=False)

    def forward(self, x, rope, mask=None):
        B, T, C = x.shape
        q = self.q_proj(x).view(B, T, self.n_heads, self.head_dim)
        k = self.k_proj(x).view(B, T, self.n_kv_heads, self.head_dim)

class TransformerBlock(nn.Module):
    def __init__(self, d_model, n_heads, n_kv_heads, ffn_dim):
        super().__init__()
        self.attn = GroupedQueryAttention(d_model, n_heads, n_kv_heads)
        self.ffn = SwiGLU(d_model, ffn_dim)   # SwiGLU beats ReLU/GELU
        self.attn_norm = RMSNorm(d_model)
        self.ffn_norm = RMSNorm(d_model)

    def forward(self, x, rope, mask=None):
        x = x + self.attn(self.attn_norm(x), rope, mask)  # Pre-norm
        x = x + self.ffn(self.ffn_norm(x))
        return x


class CppLLM(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.embed = nn.Embedding(config.vocab_size, config.d_model)
        self.rope = RotaryEmbedding(config.head_dim, config.max_seq_len)
        self.layers = nn.ModuleList([
            TransformerBlock(config.d_model, config.n_heads,
                           config.n_kv_heads, config.ffn_dim)
            for _ in range(config.n_layers)
        ])
        self.norm = RMSNorm(config.d_model)
        self.lm_head = nn.Linear(config.d_model, config.vocab_size, bias=False)
        # Weight tying - embed and lm_head share weights
        self.lm_head.weight = self.embed.weight

    def forward(self, input_ids):
        x = self.embed(input_ids)
        rope = self.rope(input_ids.shape[1], input_ids.device)
        for layer in self.layers:
            x = layer(x, rope)
        return self.lm_head(self.norm(x))
```

**Modern choices baked in:**
- ✅ RMSNorm (faster than LayerNorm)
- ✅ RoPE positional embeddings (better long context)
- ✅ SwiGLU activation (better than GELU)
- ✅ Grouped Query Attention (faster inference)
- ✅ No bias in linear layers (cleaner, standard now)
- ✅ Pre-normalization (more stable training)
